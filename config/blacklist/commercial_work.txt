# Vocabulario de OFERTA de empleo / reclutamiento (detector commercial_ad).
# Es el lado del que OFRECE trabajo (patrón spam), no el de quien lo busca.
# Un patrón por línea (palabra o regex). # = comentario. Usa (?:...) no (...).
vacantes?
puestos?\s+disponibles?
sueldo
salario
contrato\s+(?:legal|estable|indefinido)
trabajo\s+(?:estable|legal|garantizado)
oportunidad(?:es)?\s+(?:laboral(?:es)?|de\s+(?:empleo|trabajo|negocio))
ofertas?\s+de\s+(?:empleo|trabajo)
empleos?\s+disponibles?
(?:trabaja|trabajo|ingresos?|gana[rs]?|dinero)\s+desde\s+(?:casa|tu\s+m[oó]vil)
gana[rs]?\s+(?:dinero|hasta\s+\d)
ingresos?\s+(?:extra|adicionales|garantizados)
estamos\s+contratando
se\s+(?:busca|necesita[n]?)\s+(?:personal|empleados?|gente|colaboradores)
trabajo\s+que\s+m[aá]s\s+te\s+interese
si\s+est[aá]s\s+buscando\s+(?:trabajo|empleo)
# Voseo: «ganá dinero», «trabajá desde casa».
gan[aá]\s+(?:dinero|plata|hasta\s+\d)
trabaj(?:a|á|ando)\s+desde\s+(?:casa|tu\s+(?:celular|m[oó]vil|casa))
# El guion del «trabajo extra» que se cierra por privado (caso María, 20-oct-2026,
# Windows 10): «Tengo un trabajo extra al que solo le dedico unas pocas horas…».
# «trabajo extra» suelto NO: «me supone un trabajo extra configurarlo» es normal.
tengo\s+un\s+(?:trabajo|empleo|ingreso|negocio)\s+extra
(?:un|mi)\s+(?:trabajo|empleo)\s+extra\s+(?:al|con\s+el|en\s+el)\s+que
# Reclutar para la red (simulación del 29-oct: «busco 5 personas serias… se gana
# entre 200 y 300 euros al día desde el móvil… escribidme por privado» no daba nada).
# «interesados», «con ganas» o «trabajar desde casa» sueltos NO: los dice cualquiera.
busco\s+(?:a\s+)?(?:\d+\s+)?(?:personas|gente|chic[oa]s|colaboradores)\s+(?:serias?|responsables?|para\s+(?:trabajar|ganar))
se\s+(?:busca|necesita[n]?)\s+\d+\s+(?:personas|chic[oa]s|colaboradores)
gana(?:r|ncias?)?\s+[^.,\n]{0,40}?desde\s+(?:el|tu)\s+(?:m[oó]vil|celular)
